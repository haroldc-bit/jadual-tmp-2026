window.getSessionLabel = function(ev) {
    if (!ev) return "";
    const initials = { 'Kuliah': '(L)', 'Amali': '(P)', 'IL': '(IL)', 'Tutorial': '(T)', 'CA': '(CA)', 'Ulangkaji': '(UK)', 'Lain-lain': '(LL)', 'Others': '(O)', 'Taklimat': '(Taklimat)', 'Penyediaan': '(Penyediaan)' };
    const base = initials[ev.type] || '(L)';
    if (ev.sessionNumber) {
        return base.replace(')', ev.sessionNumber + ')');
    }
    return base;
};

                                                                                <p><strong className="opacity-70">Silibus:</strong> {ev.topic || 'Tiada'}</p>
                                                                                <p><strong className="opacity-70">Pengajar:</strong> {ev.instructor || 'Tiada'}</p>
                                                                            </div>
                                                                        )}

                                                                        {!(ev.type === 'IL' && ev.isBooked) && (
                                                                            <div className="mt-auto font-black opacity-60 text-[8px] uppercase">{ev.instructor}</div>
                                                                        )}
                                                                    </React.Fragment>
                                                                ) : "+"}
                                                            </div>
                                                        </td>
                                                    );
                                                })}
                                            </tr>
                                        );
                                    });
                                })}
                            </tbody>
                        </table>
                    </div>
                </div>
            </main>

            {isModalOpen && (
                <div className="fixed inset-0 z-[600] flex items-center justify-center p-4 bg-slate-900/80 backdrop-blur-sm text-black">
                    <div className="bg-white rounded-[2.5rem] shadow-2xl w-full max-w-lg p-8 border-4 border-blue-600">
                        <h3 className="font-black text-base uppercase mb-4">Kemas Kini Slot Waktu</h3>
                        <div className="space-y-4">
                            <div>
                                <label className="text-xs font-black text-slate-400 uppercase">Jenis Sesi</label>
                                <select value={formData.type} onChange={e => setFormData({...formData, type: e.target.value})} className="w-full p-3 bg-slate-50 border rounded-xl font-bold">
                                    {sessionTypes.map(t => <option key={t.value} value={t.value}>{t.label}</option>)}
                                </select>
                            </div>

                            {formData.type === 'IL' && (
                                <div className="bg-rose-50 border border-rose-200 p-4 rounded-xl space-y-3">
                                    <label className="flex items-center gap-2 font-black text-xs text-rose-900 cursor-pointer">
                                        <input type="checkbox" checked={formData.isBooked} onChange={e => setFormData({...formData, isBooked: e.target.checked})} className="w-4 h-4" />
                                        Tanda jika slot IL ini di-booking oleh Kursus
                                    </label>
                                    {formData.isBooked && (
                                        <div className="grid grid-cols-2 gap-3">
                                            <select value={formData.bookedCourse} onChange={e => setFormData({...formData, bookedCourse: e.target.value})} className="p-2 bg-white border rounded-lg font-bold text-xs">
                                                <option value="">Pilih Kod Kursus</option>
                                                {courseCodes.filter(c => c !== 'Lain-lain').map(c => <option key={c} value={c}>{c}</option>)}
                                            </select>
                                            <select value={formData.bookedPurpose} onChange={e => setFormData({...formData, bookedPurpose: e.target.value})} className="p-2 bg-white border rounded-lg font-bold text-xs">
                                                <option value="Amali">Amali</option>
                                                <option value="Ulangkaji">Ulangkaji</option>
                                                <option value="Perbincangan">Perbincangan</option>
                                            </select>
                                        </div>
                                    )}
                                </div>
                            )}

                            <div>
                                <label className="text-xs font-black text-slate-400 uppercase">Tajuk Silibus Pembelajaran</label>
                                {window.courseTopics[formData.courseCode] && (
                                    <select onChange={e => setFormData({...formData, topic: e.target.value})} className="w-full p-2 mb-2 bg-emerald-50 border border-emerald-300 rounded-lg text-xs font-bold">
                                        <option value="">-- PILIH TAJUK SILIBUS --</option>
                                        {window.courseTopics[formData.courseCode].map((t, idx) => <option key={idx} value={t}>{t}</option>)}
                                    </select>
                                )}
                                <textarea value={formData.topic} onChange={e => setFormData({...formData, topic: e.target.value})} className="w-full p-3 bg-slate-50 border rounded-xl text-xs font-bold" rows="2" placeholder="Huraian manual..." />
                            </div>

                            <div>
                                <label className="text-xs font-black text-slate-400 uppercase">Pengajar</label>
                                <select value={formData.instructor} onChange={e => setFormData({...formData, instructor: e.target.value})} className="w-full p-3 bg-slate-50 border rounded-xl font-bold">
                                    <option value="">Pilih Pengajar</option>
                                    {instructorList.map(n => <option key={n} value={n}>{n}</option>)}
                                </select>
                            </div>
                        </div>
                        <div className="flex gap-2 mt-6">
                            <button onClick={() => setIsModalOpen(false)} className="flex-1 py-3 bg-slate-100 rounded-xl font-black text-xs uppercase">Batal</button>
                            {editingEvent && userRole === 'superuser' && (
                                <button onClick={handleDelete} className="px-4 py-3 bg-red-100 text-red-600 rounded-xl font-black text-xs uppercase">Padam</button>
                            )}
                            <button onClick={handleSave} className="flex-[2] py-3 bg-blue-600 text-white rounded-xl font-black text-xs uppercase">Simpan</button>
                        </div>
                    </div>
                </div>
            )}
        </div>
    );
};

const root = ReactDOM.createRoot(document.getElementById('root'));
